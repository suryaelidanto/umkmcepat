Create `src/lib/email.ts`:

```ts
type SendEmailInput = { html?: string; subject: string; text?: string; to: string };
type SendEmailResult = { error?: string; success: boolean };

const isProduction = process.env.NODE_ENV === "production";

export async function sendEmail(input: SendEmailInput): Promise<SendEmailResult> {
  const apiKey = process.env.RESEND_API_KEY;

  if (!apiKey) {
    if (isProduction) {
      throw new Error("RESEND_API_KEY is required to send email in production.");
    }
    console.log("[email] mock", { subject: input.subject, to: input.to });
    return { success: true };
  }

  const base = process.env.RESEND_BASE_URL?.replace(/\/+$/, "") || "https://api.resend.com";
  const from = process.env.RESEND_FROM_EMAIL;
  if (!from) {
    return { error: "RESEND_FROM_EMAIL belum dikonfigurasi.", success: false };
  }

  try {
    const response = await fetch(`${base}/emails`, {
      body: JSON.stringify({ from, html: input.html, subject: input.subject, text: input.text, to: input.to }),
      headers: {
        "Authorization": `Bearer ${apiKey}`,
        "Content-Type": "application/json",
      },
      method: "POST",
    });
    if (!response.ok) {
      const error = await response.text();
      console.error("[email] Resend error:", error);
      return { error: "Gagal mengirim email. Coba lagi.", success: false };
    }
    return { success: true };
  } catch (error) {
    console.error("[email] Resend error:", error);
    return { error: "Gagal mengirim email. Coba lagi.", success: false };
  }
}
```

const isProduction = process.env.NODE_ENV === "production";
const OTPSPACE_ENDPOINT = "https://api.otpspace.com/v1/send";

export async function sendOtpViaSms(
  phone: string,
  code: string,
): Promise<{ error?: string; success: boolean }> {
  const apiKey = process.env.OTP_SPACE_API_KEY;

  if (!apiKey) {
    if (isProduction) {
      throw new Error("OTP_SPACE_API_KEY is required to send OTP in production.");
    }
    console.log("[otp] mock", { code, phone });
    return { success: true };
  }

  try {
    const response = await fetch(OTPSPACE_ENDPOINT, {
      body: JSON.stringify({ app_name: "UMKM Cepat", phone }),
      headers: { Authorization: `Bearer ${apiKey}`, "Content-Type": "application/json" },
      method: "POST",
    });
    if (!response.ok) {
      const error = await response.text();
      console.error("[otp] OTP Space error:", error);
      return { error: "Gagal mengirim OTP. Coba lagi.", success: false };
    }
    return { success: true };
  } catch (error) {
    console.error("[otp] OTP Space error:", error);
    return { error: "Gagal mengirim OTP. Coba lagi.", success: false };
  }
}
```

Note: the verified-live request body was `{phone, app_name}` (no `message`/`channel`). The OTP *message* ("Kode verifikasi…") is no longer sent as a field — OTPSpace generates the WhatsApp message from `app_name` + its template. Confirm at impl against the live API (the test that succeeded used `{phone, app_name}`).

const REQUIRED_IN_PRODUCTION = ["OTP_SPACE_API_KEY", "RESEND_API_KEY"] as const;

export function assertProvidersForProduction(): void {
  if (process.env.NODE_ENV !== "production") {
    return;
  }
  const missing = REQUIRED_IN_PRODUCTION.filter((name) => !process.env[name]);
  if (missing.length) {
    throw new Error(
      `Production requires these provider keys: ${missing.join(", ")}. Set them or mock mode stays off in prod.`,
    );
  }
}
```
